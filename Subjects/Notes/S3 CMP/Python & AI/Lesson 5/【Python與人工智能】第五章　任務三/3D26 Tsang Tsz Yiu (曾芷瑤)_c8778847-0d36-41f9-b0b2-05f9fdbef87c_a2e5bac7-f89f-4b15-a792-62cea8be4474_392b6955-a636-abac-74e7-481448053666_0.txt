import tkinter as tk
from tkinter import messagebox
import random
import time

class SuperTicTacToe:
    def __init__(self, root):
        self.root = root
        self.root.title("Task3_3X99 - 極速連擊版")
        self.root.geometry("400x600")
        self.root.configure(bg="#1a1a1a") # 深色酷炫背景
        
        # 遊戲數據
        self.player_score = 0
        self.combo = 0
        self.current_player = "X"
        self.board = [["" for _ in range(3)] for _ in range(3)]
        self.buttons = [[None for _ in range(3)] for _ in range(3)]
        
        self.create_widgets()
        self.reset_game()

    def create_widgets(self):
        # 標題與計分板
        self.header = tk.Label(
            self.root, text="3X99 究極挑戰", 
            font=('Verdana', 24, 'bold'), fg="#00ffcc", bg="#1a1a1a", pady=10
        )
        self.header.pack()

        self.score_label = tk.Label(
            self.root, text=f"得分: {self.player_score}  |  連擊: {self.combo}", 
            font=('Arial', 14), fg="white", bg="#1a1a1a"
        )
        self.score_label.pack()

        # 棋盤容器
        board_frame = tk.Frame(self.root, bg="#333333", padx=10, pady=10, relief="sunken", bd=5)
        board_frame.pack(pady=20)

        for r in range(3):
            for c in range(3):
                btn = tk.Button(
                    board_frame, text="", font=('Impact', 36),
                    width=3, height=1, bg="#2d2d2d", fg="white",
                    activebackground="#404040", relief="flat",
                    command=lambda row=r, col=c: self.player_action(row, col)
                )
                btn.grid(row=r, column=c, padx=5, pady=5)
                self.buttons[r][c] = btn

        # 狀態列
        self.status_label = tk.Label(
            self.root, text="準備好了嗎？先手出擊！", 
            font=('Microsoft JhengHei', 12), fg="#aaaaaa", bg="#1a1a1a"
        )
        self.status_label.pack(side="bottom", pady=20)

    def player_action(self, r, col):
        if self.board[r][col] == "" and self.current_player == "X":
            self.root.bell() # 點擊音效
            self.make_move(r, col, "X", "#00d4ff") # 酷藍色
            
            if not self.check_end():
                self.current_player = "O"
                self.status_label.config(text="電腦計算中...", fg="#ff4444")
                # 模擬電腦思考
                self.root.after(400, self.ai_action)

    def ai_action(self):
        # 使用 Minimax 確保無敵
        move = self.get_best_move()
        if move:
            self.make_move(move[0], move[1], "O", "#ff0055") # 霓虹紅
            if not self.check_end():
                self.current_player = "X"
                self.status_label.config(text="輪到你了！快！", fg="#00ffcc")

    def make_move(self, r, col, char, color):
        self.board[r][col] = char
        btn = self.buttons[r][col]
        btn.config(text=char, fg=color, state="disabled", disabledforeground=color)
        
        # 閃爍動畫特效
        original_bg = btn.cget("bg")
        btn.config(bg="white")
        self.root.after(100, lambda: btn.config(bg=original_bg))

    def get_best_move(self):
        best_score = -float('inf')
        move = None
        for r in range(3):
            for c in range(3):
                if self.board[r][c] == "":
                    self.board[r][c] = "O"
                    score = self.minimax(self.board, 0, False)
                    self.board[r][c] = ""
                    if score > best_score:
                        best_score = score
                        move = (r, c)
        return move

    def minimax(self, board, depth, is_max):
        res = self.evaluate()
        if res == "O": return 10 - depth
        if res == "X": return depth - 10
        if all(board[r][c] != "" for r in range(3) for c in range(3)): return 0

        if is_max:
            best = -float('inf')
            for r in range(3):
                for c in range(3):
                    if board[r][c] == "":
                        board[r][c] = "O"; s = self.minimax(board, depth+1, False); board[r][c] = ""; best = max(s, best)
            return best
        else:
            best = float('inf')
            for r in range(3):
                for c in range(3):
                    if board[r][c] == "":
                        board[r][c] = "X"; s = self.minimax(board, depth+1, True); board[r][c] = ""; best = min(s, best)
            return best

    def evaluate(self):
        for i in range(3):
            if self.board[i][0] == self.board[i][1] == self.board[i][2] != "": return self.board[i][0]
            if self.board[0][i] == self.board[1][i] == self.board[2][i] != "": return self.board[0][i]
        if self.board[0][0] == self.board[1][1] == self.board[2][2] != "": return self.board[0][0]
        if self.board[0][2] == self.board[1][1] == self.board[2][0] != "": return self.board[0][2]
        return None

    def check_end(self):
        winner = self.evaluate()
        if winner:
            if winner == "X":
                self.player_score += 100; self.combo += 1
                messagebox.showinfo("VICTORY", "天啊！你擊敗了無敵 AI！")
            else:
                self.combo = 0
                messagebox.showerror("DEFEAT", "被電腦虐了... 嫩！")
            self.reset_game()
            return True
        if all(self.board[r][c] != "" for r in range(3) for c in range(3)):
            self.player_score += 10 # 平手也給點分
            messagebox.showinfo("DRAW", "平手！你的防禦很完美。")
            self.reset_game()
            return True
        return False

    def reset_game(self):
        self.score_label.config(text=f"得分: {self.player_score}  |  連擊: {self.combo}")
        self.current_player = "X"
        for r in range(3):
            for c in range(3):
                self.board[r][c] = ""
                self.buttons[r][c].config(text="", state="normal", bg="#2d2d2d")

if __name__ == "__main__":
    root = tk.Tk()
    app = SuperTicTacToe(root)
    root.mainloop()
